cmake_minimum_required(VERSION 3.22.1)
project(girassol_projection LANGUAGES CXX)
add_library(girassol_memory OBJECT mind/vector_memory.cpp)
target_include_directories(girassol_memory SYSTEM PRIVATE third_party/hnswlib)
target_compile_features(girassol_memory PRIVATE cxx_std_20)
target_compile_options(girassol_memory PRIVATE -fno-exceptions -Wall -Wextra -Werror)
target_compile_definitions(girassol_memory PRIVATE HNSWLIB_ENABLE_EXCEPTIONS=OFF HNSWLIB_USE_PREFETCH=0)
set_target_properties(girassol_memory PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(girassol_projection SHARED projection.cpp projection_jni.cpp city_mesh.cpp city_mesh_jni.cpp audio/synth.cpp audio/audio_jni.cpp mind/plasticity.cpp mind/mind_jni.cpp mind/vector_memory_jni.cpp evolution/cma_es.cpp evolution/cma_jni.cpp $<TARGET_OBJECTS:girassol_memory>)
target_compile_features(girassol_projection PRIVATE cxx_std_20)
target_compile_options(girassol_projection PRIVATE -Wall -Wextra -Werror)
target_link_libraries(girassol_projection PRIVATE aaudio)

include(cognition/llama-dependency.cmake)
add_library(girassol_llama SHARED cognition/inference.cpp cognition/llama_jni.cpp)
target_compile_features(girassol_llama PRIVATE cxx_std_20)
target_compile_options(girassol_llama PRIVATE -Wall -Wextra -Werror)
target_link_libraries(girassol_llama PRIVATE llama)

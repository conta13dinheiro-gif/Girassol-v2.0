cmake_minimum_required(VERSION 3.22.1)
project(girassol_cognition_test LANGUAGES C CXX)
set(NATIVE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../main/cpp")
include("${NATIVE_ROOT}/cognition/llama-dependency.cmake")
add_executable(cognition_test "${NATIVE_ROOT}/cognition/inference.cpp" cognition_test.cpp)
target_compile_features(cognition_test PRIVATE cxx_std_20)
target_compile_options(cognition_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(cognition_test PRIVATE llama)

option(GIRASSOL_TEST_JNI "Build production JNI adapter for host JVM smoke tests" OFF)
if(GIRASSOL_TEST_JNI)
    find_package(JNI REQUIRED)
    add_library(girassol_llama SHARED "${NATIVE_ROOT}/cognition/inference.cpp" "${NATIVE_ROOT}/cognition/llama_jni.cpp")
    target_compile_features(girassol_llama PRIVATE cxx_std_20)
    target_compile_options(girassol_llama PRIVATE -Wall -Wextra -Werror)
    target_include_directories(girassol_llama PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(girassol_llama PRIVATE llama)
endif()
